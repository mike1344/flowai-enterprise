### 3.2 复跑命令

```powershell
[Console]::OutputEncoding=[System.Text.Encoding]::UTF8
$files = @(Get-ChildItem -Recurse -File 'F:\Y1\docs' -Filter *.md) + @(Get-Item 'F:\Y1\README.md')
$bad = 0; $total = 0
foreach ($f in $files) {
  $text = Get-Content -Raw -Encoding UTF8 $f.FullName
  foreach ($m in [regex]::Matches($text, '\]\(([^)#\s]+?)(#[^)]*)?\)')) {
    $t = $m.Groups[1].Value
    if ($t -match '^(https?:|mailto:)') { continue }
    $total++
    if (-not (Test-Path -LiteralPath (Join-Path $f.DirectoryName $t))) { $bad++; "BROKEN $($f.Name) -> $t" }
  }
}
"链接总数：$total；断链：$bad"
```
